  ```python
  """F-002 优化流程状态机（唯一实现，MUST-07）；乐观锁串行化（MUST-11）。"""

  from __future__ import annotations

  import uuid
  from enum import Enum

  from devprompt_ai.domain.errors import StateError
  from devprompt_ai.domain.models import FlowSession, SessionStore


  class FlowState(str, Enum):
      INIT = "INIT"
      RETRIEVING = "RETRIEVING"
      PROBING = "PROBING"
      ASKING = "ASKING"
      CONFIRMING = "CONFIRMING"
      GENERATING = "GENERATING"
      RANKING = "RANKING"
      DONE = "DONE"
      ABORTED = "ABORTED"


  # GENERATING→RANKING 为阶段五推荐接入预留；阶段四单套走 GENERATING→DONE。
  # RETRIEVING→ASKING：F-102 候选 <3 条（manual_required）直达反问。
  TRANSITIONS: dict[FlowState, frozenset[FlowState]] = {
      FlowState.INIT: frozenset({FlowState.RETRIEVING, FlowState.ABORTED}),
      FlowState.RETRIEVING: frozenset({FlowState.PROBING, FlowState.ASKING, FlowState.ABORTED}),
      FlowState.PROBING: frozenset({FlowState.ASKING, FlowState.GENERATING, FlowState.ABORTED}),
      FlowState.ASKING: frozenset({FlowState.CONFIRMING, FlowState.GENERATING, FlowState.ABORTED}),
      FlowState.CONFIRMING: frozenset({FlowState.ASKING, FlowState.GENERATING, FlowState.ABORTED}),
      FlowState.GENERATING: frozenset({FlowState.RANKING, FlowState.DONE, FlowState.ABORTED}),
      FlowState.RANKING: frozenset({FlowState.DONE, FlowState.ABORTED}),
      FlowState.DONE: frozenset(),
      FlowState.ABORTED: frozenset(),
  }


  def new_session_id() -> str:
      return uuid.uuid4().hex


  def validate_transition(current: FlowState, target: FlowState) -> None:
      if target not in TRANSITIONS[current]:
          raise StateError(
              f"非法状态转移 {current.value} -> {target.value}",
              code=6001,

      session.state = target.value
      if not await store.save(session):
          raise StateError(
              "会话版本冲突，拒绝写入（乐观锁）",
              code=6001,
              context={"session_id": session.session_id},
          )


  async def load_or_create(
      store: SessionStore, user_id: str, session_id: str | None = None
  ) -> FlowSession:
      if session_id is not None:
          session = await store.load(session_id)
          if session is None:
              raise StateError("流程 ID 不存在", code=6003, context={"session_id": session_id})
          return session
      session = FlowSession(session_id=new_session_id(), user_id=user_id, state=FlowState.INIT.value)
      await store.create(session)
      return session
  ```

  ```python
  """F-501 L1 会话存储（SQLite，WAL）：CAS 乐观锁写入（MUST-11）。SQL 限于 memory/（MUST-06）。"""

  from __future__ import annotations

  import asyncio
  import json
  import sqlite3
  from pathlib import Path

  from devprompt_ai.domain.models import FlowSession
  from devprompt_ai.memory.schema import SESSIONS_DDL

  _BUSY_TIMEOUT_MS = 5000


  class SQLiteSessionStore:
      """每操作独立连接 + asyncio.to_thread（sqlite3 同步 API 的异步包装，MUST-13）。"""

      def __init__(self, db_path: str) -> None:
          self._db_path = str(Path(db_path).expanduser())

      def _connect(self) -> sqlite3.Connection:
          Path(self._db_path).parent.mkdir(parents=True, exist_ok=True)
          conn = sqlite3.connect(self._db_path, timeout=_BUSY_TIMEOUT_MS / 1000)
          conn.execute(f"PRAGMA busy_timeout = {_BUSY_TIMEOUT_MS}")
          return conn

      async def init_schema(self) -> None:
          def _run() -> None:
              with self._connect() as conn:
                  conn.execute("PRAGMA journal_mode = WAL")
                  conn.executescript(SESSIONS_DDL)
          await asyncio.to_thread(_run)

      async def create(self, session: FlowSession) -> None:
          def _run() -> None:
              with self._connect() as conn:
                  conn.execute(
                      "INSERT INTO forge_sessions (session_id, user_id, state, version, payload)"
                      " VALUES (?, ?, ?, ?, ?)",
                      (session.session_id, session.user_id, session.state, session.version,
                       json.dumps(session.data, ensure_ascii=False)),
                  )
          await asyncio.to_thread(_run)

      async def load(self, session_id: str) -> FlowSession | None:
          def _run() -> FlowSession | None:
              with self._connect() as conn:
                  row = conn.execute(
                      "SELECT user_id, state, version, payload FROM forge_sessions"
                      " WHERE session_id = ?",
                      (session_id,),
                  ).fetchone()
              if row is None:
                  return None
              return FlowSession(session_id=session_id, user_id=row[0], state=row[1],
                                 version=row[2], data=json.loads(row[3]))
          return await asyncio.to_thread(_run)

      async def save(self, session: FlowSession) -> bool:
          def _run() -> bool:
              with self._connect() as conn:
                  cur = conn.execute(
                      "UPDATE forge_sessions SET state = ?, version = version + 1, payload = ?"
                      " WHERE session_id = ? AND version = ?",
                      (session.state, json.dumps(session.data, ensure_ascii=False),
                       session.session_id, session.version),
                  )
                  if cur.rowcount == 0:
                      return False
              session.version += 1
              return True
          return await asyncio.to_thread(_run)
  ```
  - 注意事项：`sqlite3.connect` 默认隔离级别开启隐式事务，`with conn` 提交；CAS 语义靠 WHERE version 实现，0 行更新即版本冲突（6001 由调用方 `state.transition` 抛出）；WAL 文件（-wal/-shm）随库文件同目录，已被 `.devprompt-ai/` gitignore 覆盖

  ```python
  from __future__ import annotations

  from dataclasses import dataclass
  from collections.abc import Sequence

  from devprompt_ai.config import AppConfig
  from devprompt_ai.domain.decision.port import FactProbeOutput
  from devprompt_ai.domain.decision.probes import PROBE_ASK_TEXT
  from devprompt_ai.domain.models import FactResult

  _EVIDENCE_LIMIT = 120


  @dataclass(frozen=True, slots=True)
  class UnknownItem:
      """具体未决事实（MUST-26 v1.1）：每问绑定一项。"""

      key: str
      reason: str
      best_guess: str
      ask: str


  def build_unknowns(
      probe: FactProbeOutput,
      facts: Sequence[FactResult],
      *,
      clarifications: dict[str, str] | None = None,
  ) -> list[UnknownItem]:
      """失败或低置信的事实 → 未知项；用户已答复的 key 视为已澄清跳过。"""
      clar = clarifications or {}
      judgment_by_key = {j.key: j for j in probe.judgments}
      items: list[UnknownItem] = []
      for fact in facts:
          if fact.key in clar:
              continue  # 已澄清（MUST-26：答复即解除该未知项）
          judgment = judgment_by_key.get(fact.key)
          if fact.passed and (judgment is None or judgment.passed):
              continue
          reason = fact.evidence if not fact.passed else (
              f"研判置信度不足（{judgment.confidence:.2f}）" if judgment else "未判定")
          items.append(UnknownItem(
              key=fact.key,
              reason=reason[:_EVIDENCE_LIMIT],
              best_guess=fact.evidence[:_EVIDENCE_LIMIT] if fact.passed else "",
              ask=PROBE_ASK_TEXT.get(fact.key, fact.key),
          ))
      return items


  @dataclass(slots=True)
  class GuardrailState:
      """MUST-26 软护栏：触发时不静默终止，由 service 转为显式用户裁决。"""

      rounds: int = 0
      cost_tokens: int = 0

      def should_trigger(self, cfg: AppConfig) -> bool:
          if self.rounds >= cfg.followup_soft_round_guard:
              return True
          return cfg.followup_cost_budget > 0 and self.cost_tokens >= cfg.followup_cost_budget

      def reset(self) -> None:
          """用户裁决「继续追问」后清零。"""
          self.rounds = 0
          self.cost_tokens = 0
  ```

  ```python
  """F-502 L2 项目记忆：Git notes（key=devprompt，单条 ≤8KB 纯文本）。"""

  from __future__ import annotations

  from collections.abc import Sequence

  from devprompt_ai.domain.retrieval.ports import CommandRunner

  NOTE_REF = "devprompt"
  MAX_NOTE_BYTES = 8192
  _GIT_OK = frozenset({0, 1})  # 1 = notes list 为空等


  def _git(root: str, *args: str, route_key: str = "") -> list[str]:
      args_list = ["git", "-C", root, *args]
      if route_key:
          # 测试路由键约定：LocalCommandRunner/测试 Fake 过滤 --route-key
          args_list.append(f"--route-key={route_key}")
      return args_list


  async def write_note(root: str, content: str, runner: CommandRunner) -> None:
      """写项目规范 note（覆盖式 add -f）。超 8KB 拒绝（F-502 行为列）。"""
      if len(content.encode("utf-8")) > MAX_NOTE_BYTES:
          raise ValueError(f"L2 note 超过 {MAX_NOTE_BYTES} 字节上限")
      await runner.run(
          _git(root, "notes", "--ref", NOTE_REF, "add", "-f", "-m", content, "HEAD", route_key="notes-add"))


  async def read_notes(root: str, runner: CommandRunner) -> list[tuple[str, str]]:
      """返回 [(对象哈希, note 内容)]；无 notes 返回空列表。"""
      listing = await runner.run(
          _git(root, "notes", "--ref", NOTE_REF, "list", route_key="notes-list"),
          allowed_exit_codes=_GIT_OK)
      notes: list[tuple[str, str]] = []
      for line in listing.splitlines():
          parts = line.split()
          if len(parts) != 2:
              continue
          obj = parts[0]
          content = await runner.run(
              _git(root, "notes", "--ref", NOTE_REF, "show", obj, route_key="notes-show"),
              allowed_exit_codes=_GIT_OK)
          notes.append((obj, content))
      return notes


  async def ensure_push_config(root: str, runner: CommandRunner) -> None:
      """F-502：推送配置使 notes 随远端同步（幂等 --add）。"""
      await runner.run(_git(root, "config", "--add", "remote.origin.push", "refs/notes/*", route_key="config-add"))


  async def refresh_notes(root: str, runner: CommandRunner) -> None:
      """手动刷新入口：拉取远端 notes（无远端时 ProcessError 上抛由调用方提示）。"""
      await runner.run(_git(root, "fetch", "origin", "refs/notes/*:refs/notes/*", route_key="fetch"))


  async def init_note(root: str, spec_text: str, runner: CommandRunner) -> None:
      """F-601 init：写入初始项目规范 note（要求 root 已有 commit）。"""
      await write_note(root, spec_text, runner)
  ```

  ```python
  @dataclass(slots=True)
  class FlowSnapshot:
      """表现层快照：CLI/Web 只渲染它（MUST-10）。"""

      session_id: str
      status: Literal["ask", "done", "guardrail"]
      state: str
      questions: list[UnknownItem] = field(default_factory=list)
      cleared: list[str] = field(default_factory=list)

  class ForgeService:
      def __init__(
          self, *, cfg: AppConfig, decider: DecisionPort, sessions: SessionStore,
          runner: CommandRunner, reader: FileReader, generator: GeneratePort,
          symbols_provider: SymbolTableProvider, symbols_override: list[Symbol] | None = None,
      ) -> None:
          self._cfg = cfg
          self._decider = decider
          self._sessions = sessions
          self._runner = runner
          self._reader = reader
          self._generator = generator
          self._symbols_provider = symbols_provider
          self._symbols_override = symbols_override

      async def optimize(self, user_id: str, project_root: str, sentence: str, *, request_id: str) -> FlowSnapshot:
          """F-001：创建/恢复会话，执行阶段一至二（检索+探针），路由 ask/done。"""
          session = await load_or_create(self._sessions, user_id)
          session.data = {
              "sentence": sentence, "project_root": project_root, "last_request_id": request_id,
              "clarifications": {}, "rounds": 0, "cost_tokens": 0,
          }
          await transition(session, FlowState.RETRIEVING, self._sessions)

          symbols = self._symbols_override
          if symbols is None:
              table = await self._symbols_provider.get(project_root)
              symbols = table.symbols
          result = build_candidates(sentence, symbols)
          session.data["candidates"] = [
              {"name": c.name, "path": c.path, "score": c.score, "source": c.source}
              for c in result.candidates
          ]

          if result.manual_required:
              session.data["unknowns"] = [UnknownItem(
                  key="manual_target",
                  reason=f"候选不足（{result.total_matched} 条）",
                  best_guess="",
                  ask="请指明目标符号名与文件路径（格式：符号@路径）").__dict__]
              session.data["level"] = "A"
              await transition(session, FlowState.ASKING, self._sessions)
              return self._snapshot(session, "ask")

          top = result.candidates[0]
          symbol = Symbol(name=top.name, path=top.path, line=top.line, kind="function")
          location = await locate_symbol(symbol, self._runner, self._reader)
          facts = await verify_facts(location, sentence, self._runner, self._reader)
          session.data["facts"] = [{"key": f.key, "passed": f.passed, "evidence": f.evidence} for f in facts]
          session.data["target"] = {"name": top.name, "path": top.path}
          await transition(session, FlowState.PROBING, self._sessions)
          return await self._probe_and_route(session, request_id=request_id)

      async def ask_followup(
          self, user_id: str, session_id: str, answers: dict[str, str], *, request_id: str
      ) -> FlowSnapshot:
          """F-204：批量答复 → 重新研判 → ask/guardrail/done。"""
          session = await self._require(user_id, session_id, {FlowState.ASKING, FlowState.CONFIRMING})
          session.data.setdefault("clarifications", {}).update(answers)
          session.data["last_request_id"] = request_id
          session.data["rounds"] = int(session.data.get("rounds", 0)) + 1
          await transition(session, FlowState.CONFIRMING, self._sessions)
          return await self._reprobe(session)

      async def escape(self, user_id: str, session_id: str, *, request_id: str) -> FlowSnapshot:
          """MUST-26 ②：用户逃生「直接生成」→ 保守版。"""

      async def _probe_and_route(self, session: FlowSession, *, request_id: str) -> FlowSnapshot:
          facts = [FactResult(**f) for f in session.data.get("facts", [])]
          probe = await self._decider.probe_facts(session.data["sentence"], facts)
          session.data["probe_confidence"] = probe.overall_confidence
          unknowns = build_unknowns(probe, facts, clarifications=session.data.get("clarifications"))
          session.data["unknowns"] = [dataclasses.asdict(u) for u in unknowns]
          facts_complete = bool(facts) and all(f.passed for f in facts) and \
              probe.overall_confidence >= self._cfg.fact_probe_threshold
          has_symbol = bool(session.data.get("target"))
          session.data["level"] = resolve_level(has_symbol=has_symbol, facts_complete=facts_complete)
          if unknowns:
              await transition(session, FlowState.ASKING, self._sessions)
              return self._snapshot(session, "ask")
          return await self._generate_and_finish(session)

      async def _reprobe(self, session: FlowSession) -> FlowSnapshot:
          """F-204：答复后重新研判（取证结果不变，clarifications 已更新）。"""
          facts = [FactResult(**f) for f in session.data.get("facts", [])]
          probe = await self._decider.probe_facts(session.data["sentence"], facts)
          unknowns = build_unknowns(probe, facts, clarifications=session.data.get("clarifications"))
          session.data["unknowns"] = [dataclasses.asdict(u) for u in unknowns]
          guard = GuardrailState(rounds=int(session.data.get("rounds", 0)),
                                 cost_tokens=int(session.data.get("cost_tokens", 0)))
          if guard.should_trigger(self._cfg):
              logger.info("followup.guardrail",
                          extra={"rounds": guard.rounds, "cost_tokens": guard.cost_tokens,
                                 "request_id": session.data.get("last_request_id", "")})
              return self._snapshot(session, "guardrail")  # 停在 CONFIRMING 等裁决（6002 语义）
          if unknowns:
              await transition(session, FlowState.ASKING, self._sessions)
              return self._snapshot(session, "ask")
          return await self._generate_and_finish(session)

      async def _generate_and_finish(self, session: FlowSession) -> FlowSnapshot:
          await transition(session, FlowState.GENERATING, self._sessions)
          ctx = GenerationContext(
              sentence=session.data["sentence"], project_root=session.data["project_root"],
              level=session.data.get("level", "A"),
              facts_summary=session.data.get("facts_summary", _summarize_facts(session)),
              clarifications=session.data.get("clarifications", {}),
              conservative=bool(session.data.get("conservative", False)),
          )
          draft = await self._generator.generate_single(ctx)
          usage = last_usage()
          if usage:
              session.data["cost_tokens"] = int(session.data.get("cost_tokens", 0)) + usage["total_tokens"]
          session.data["variant"] = {"title": draft.title, "body": draft.body,
                                     "changes": draft.changes, "level": draft.level}
          await transition(session, FlowState.DONE, self._sessions)  # RANKING 阶段五接入
          return self._snapshot(session, "done")

      def _snapshot(self, session: FlowSession, status: str) -> FlowSnapshot:
          questions = [UnknownItem(**u) for u in session.data.get("unknowns", [])]
          return FlowSnapshot(
              session_id=session.session_id, status=status, state=session.state,
              questions=questions, cleared=sorted(session.data.get("clarifications", {}).keys()),
              variant=VariantDraft(**session.data["variant"]) if "variant" in session.data else None,
          )

      async def _require(
          self, user_id: str, session_id: str, allowed: set[FlowState]
      ) -> FlowSession:
          session = await self._sessions.load(session_id)
          if session is None or session.user_id != user_id:
              raise StateError("流程 ID 不存在", code=6003, context={"session_id": session_id})
          if FlowState(session.state) not in allowed:
              raise StateError(
                  f"当前状态 {session.state} 不允许该操作", code=6001,
                  context={"session_id": session_id, "state": session.state},
              )
          return session


  def _summarize_facts(session: FlowSession) -> str:
      parts = [f"{f['key']}={'真' if f['passed'] else '假'}（{f['evidence'][:80]}）"
               for f in session.data.get("facts", [])]
      target = session.data.get("target", {})
      return f"目标：{target.get('name', '')}@{target.get('path', '')}；" + "；".join(parts)


  async def build_default(cfg: AppConfig) -> ForgeService:

      from devprompt_ai.harness.llm import LLMGenerator

      sessions = SQLiteSessionStore(cfg.db_path)
      await sessions.init_schema()

          cfg=cfg, decider=create_decider(cfg), sessions=sessions,
          runner=LocalCommandRunner(), reader=LocalFileReader(),
          generator=LLMGenerator(cfg), symbols_provider=SymbolTableProvider(LocalCommandRunner()),
      )
  ```

  ```python
  async def drive_optimize(
      console: Console, *, sentence: str, project: str, json_out: bool
  ) -> None:
      cfg = AppConfig()
      cfg.validate()
      service = await build_default(cfg)
      snapshot = await service.optimize("local", project, sentence, request_id=uuid.uuid4().hex)
      while snapshot.status in ("ask", "guardrail"):
          if json_out:
              print(json.dumps(_payload(snapshot), ensure_ascii=False))
          if snapshot.status == "guardrail":
              choice = prompts_flow.ask_guardrail(console, snapshot)
              snapshot = await service.guardrail_confirm(
                  "local", snapshot.session_id, proceed=(choice == "continue"),
                  request_id=uuid.uuid4().hex)
          else:
              render.print_questions(console, snapshot)
              answers = prompts_flow.collect_answers(console, snapshot.questions)
              if answers.get("__escape__"):
                  snapshot = await service.escape("local", snapshot.session_id,
                                                  request_id=uuid.uuid4().hex)

                  answers.pop("__escape__", None)
                  snapshot = await service.ask_followup(
                      "local", snapshot.session_id, answers, request_id=uuid.uuid4().hex)

      else:
  ```

  ```python
  """F-602 反问交互：批量问答、逃生口、护栏裁决面板。"""

  from __future__ import annotations

  from rich.console import Console

  from devprompt_ai.service import FlowSnapshot

  ESCAPE = "!"


  def collect_answers(console: Console, questions: list) -> dict[str, str]:
      """收集批量答复；任一轮输入 ! 即逃生（MUST-26 ②）。空回车 = 跳过（采用推断）。"""
      console.print("[bold]请逐项回答（回车=采用推断；输入 [yellow]![/yellow] = 直接生成）[/bold]")
      answers: dict[str, str] = {}
      for q in questions:
          value = console.input(f"  {q.key} > ").strip()
          if value == ESCAPE:
              return {"__escape__": "1"}
          if value:
              answers[q.key] = value
      return answers


  def ask_guardrail(console: Console, snapshot: FlowSnapshot) -> str:
      """MUST-26 ③：软护栏裁决面板。返回 continue / generate。"""
      cleared = ", ".join(snapshot.cleared) or "（无）"
      console.print(Panel.fit(
          f"已追问多轮。已澄清：{cleared}\n请选择：[bold]c[/bold] = 继续追问（清零护栏）｜[bold]g[/bold] = 直接生成",
          title="软护栏", border_style="yellow"))
      while True:
          choice = console.input("  c/g > ").strip().lower()
          if choice in ("c", "g"):
              return "continue" if choice == "c" else "generate"
  ```
