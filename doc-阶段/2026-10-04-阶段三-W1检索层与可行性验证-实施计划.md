  ```python
  _CJK_RUN = re.compile(r"[\u3400-\u4dbf\u4e00-\u9fff\uf900-\ufaff]+")
  # 连续 ASCII 词段（字母数字）
  _ASCII_RUN = re.compile(r"[A-Za-z0-9]+")
  # 驼峰边界：小写/数字→大写，或 大写串→大写+小写（HTTPServer → HTTP|Server）
  _CAMEL_SPLIT = re.compile(
      r"[A-Z]+(?=[A-Z][a-z0-9])|[A-Z]?[a-z0-9]+|[A-Z]+"
  )


  def split_identifier(name: str) -> list[str]:
      """将标识符/路径拆为小写词元：驼峰、下划线、连字符、点、斜杠均为边界。

      例：SubmitButton -> [submit, button]；src/v2/UserList.tsx -> [src, v2, user, list, tsx]
      """
      words: list[str] = []
      for raw in _ASCII_RUN.findall(name):
          words.extend(w.lower() for w in _CAMEL_SPLIT.findall(raw) if w)
      return words


  def cjk_bigrams(text: str) -> list[str]:
      """将文本中的连续中文段切为相邻二元组；单字段保留单字。"""
      grams: list[str] = []
      for run in _CJK_RUN.findall(text):
          if len(run) == 1:
              grams.append(run)
          else:
              grams.extend(run[i : i + 2] for i in range(len(run) - 1))
      return grams


  def tokenize_sentence(sentence: str) -> list[str]:
      """从原句提取匹配词元：英文词元 + 中文二元组，去重保序。

      例："把用户中心页面的提交按钮背景色改为 primary-500"
        -> [用户, 中心, 页面, 面的, 提交, 交按, 按钮, 背, 背景, 景色, 色改, 改为, primary, 500]
      """
      terms: list[str] = []
      for word in split_identifier(sentence):
          if word not in terms:
              terms.append(word)
      for gram in cjk_bigrams(sentence):
          if gram not in terms:
              terms.append(gram)
      return terms
  ```

  ```python
  """哈希纯工具：原句 sha256、mtime 摘要（MUST-14 缓存键的基元）。

  mtime_digest 只做纯计算——路径与 mtime 由调用方收集（显式传入，无隐式 I/O，§4.7）。
  """

  from __future__ import annotations

  import hashlib


  def sha256_hex(data: str) -> str:
      return hashlib.sha256(data.encode("utf-8")).hexdigest()


  def mtime_digest(entries: list[tuple[str, int]]) -> str:
      """将 (路径, mtime_ns) 列表排序后拼接待摘要串，输出 sha256 前 16 位。

      排序保证与传入顺序无关；mtime 变化必然改变摘要（MUST-14 失效条件）。
      """
      lines = "\n".join(f"{path}|{mtime}" for path, mtime in sorted(entries))
      return sha256_hex(lines)[:16]
  ```

  ```python
  """异步子进程执行封装（MUST-13 的唯一通道）。

  规则：一律 asyncio.create_subprocess_exec；禁 shell=True；必须设超时（默认 10 秒）；
  超时/异常路径 kill() + await wait() 防僵尸进程；输出统一 UTF-8 errors=replace。
  """

  from __future__ import annotations

  import asyncio

  DEFAULT_TIMEOUT = 10.0
  _ENCODING = "utf-8"

  DEFAULT_ALLOWED_EXIT_CODES = frozenset({0})


  class ProcessError(RuntimeError):
      """子进程非零退出（且退出码不在 allowed_exit_codes 内）。"""

      def __init__(self, args: list[str], code: int, stderr: str) -> None:
          super().__init__(f"命令失败 exit={code}: {' '.join(args)}\nstderr: {stderr[:500]}")
          self.code = code


  class ProcessTimeoutError(RuntimeError):
      """子进程超时（已 kill）。"""


  async def run_command(
      args: list[str],
      *,
      timeout: float = DEFAULT_TIMEOUT,
      allowed_exit_codes: frozenset[int] = DEFAULT_ALLOWED_EXIT_CODES,
      cwd: str | None = None,
  ) -> str:
      """执行外部命令并返回 stdout 文本。exit 码不在允许集时抛 ProcessError。"""
      proc = await asyncio.create_subprocess_exec(
          *args,
          cwd=cwd,
          stdout=asyncio.subprocess.PIPE,
          stderr=asyncio.subprocess.PIPE,
      )
      try:
          stdout_b, stderr_b = await asyncio.wait_for(proc.communicate(), timeout=timeout)
      except (TimeoutError, asyncio.TimeoutError):
          proc.kill()
          await proc.wait()  # 僵尸进程清理（MUST-13）
          raise ProcessTimeoutError(f"命令超时({timeout}s): {' '.join(args)}") from None

      stdout = stdout_b.decode(_ENCODING, errors="replace")
      stderr = stderr_b.decode(_ENCODING, errors="replace")
      code = proc.returncode if proc.returncode is not None else -1
      if code not in allowed_exit_codes:
          raise ProcessError(args, code, stderr)
      return stdout
  ```

  - 注意事项：Python 3.12 中 `asyncio.wait_for` 抛 `TimeoutError`（内建）与 `asyncio.TimeoutError` 已合一，except 双捕兼容 3.10

  ```python
      """解析 ctags --output-format=json 的逐行 JSON 输出；脏行与非 tag 行跳过。"""
      symbols: list[Symbol] = []
      for line in raw.splitlines():
          line = line.strip()
          if not line:
              continue
          try:
              obj = json.loads(line)
          except json.JSONDecodeError:
              continue
          if not isinstance(obj, dict) or obj.get("_type") != "tag":
              continue
          name = obj.get("name")
          path = obj.get("path")
          raw_line = obj.get("line")
          if not name or not path or not isinstance(raw_line, int):
              continue
          kind = _KIND_MAP.get(str(obj.get("kind", "")), "other")
          # 路径归一化为 POSIX 相对形式（ctags 在 Windows 可能输出反斜杠）
          symbols.append(
              Symbol(
                  name=str(name),
                  path=str(path).replace("\\", "/").lstrip("./"),
                  line=raw_line,
                  kind=kind,
              )
          )
      return symbols


  def cache_key(project_root: str, mtimes: list[tuple[str, int]]) -> str:
      """符号表缓存键 = 项目绝对路径 + mtime 摘要（MUST-14，TTL 1 小时）。"""
      return sha256_hex(f"{project_root}\n{sorted(mtimes)}")[:16]


  def build_ctags_args(project_root: str) -> list[str]:
      """组装 ctags 命令：JSON 输出 + 行号 + 排除目录（纯函数便于测试断言）。"""
      args = ["ctags", "-f", "-", "--output-format=json", "--fields=+n", "-R"]
      for d in EXCLUDED_DIRS:
          args.append(f"--exclude={d}")
      args.append(project_root)
      return args
  ```

  ```python
  # Protocol 实现（MUST-06 落点：domain/retrieval/ports.py 的实现在此）
  # ---------------------------------------------------------------------------

  import os
  from collections.abc import Sequence

  from devprompt_ai.common.hashing import mtime_digest

  )

  SYMBOL_TABLE_TTL = 3600.0  # MUST-14：符号表缓存 TTL 1 小时


  class LocalCommandRunner:
      """CommandRunner 的本地实现：转发 common.process.run_command。"""

      async def run(
          self,
          args: list[str],
          *,
          timeout: float = 10.0,
          allowed_exit_codes: frozenset[int] = frozenset({0}),
          cwd: str | None = None,
      ) -> str:

              args, timeout=timeout, allowed_exit_codes=allowed_exit_codes, cwd=cwd
          )


  class LocalFileReader:
      """FileReader 的本地实现（路径规范化由调用方在 domain 层完成）。"""

      async def read(self, path: str) -> str:
          with open(path, encoding="utf-8", errors="replace") as f:
              return f.read()

      async def exists(self, path: str) -> bool:
          return os.path.isfile(path)


  class SymbolTableProvider:
      """F-101 编排：ctags 调用 + 解析 + 进程内缓存（键含 mtime 摘要，TTL 1h）。

      增量策略采用 SHOULD-04「惰性优于定时」：访问时以 mtime 摘要探测变更，
      变更即整表重建（符号表重建成本低，不做逐文件 diff）。
      """

      def __init__(self, runner: object, ttl_seconds: float = SYMBOL_TABLE_TTL) -> None:
          self._runner = runner
          self._ttl = ttl_seconds
          self._cache_key: str | None = None
          self._cache_symbols: list[Symbol] | None = None
          self._cached_at: float = 0.0
          self.mtime_digest_override: str | None = None  # 测试注入点
          self.calls_args: list[list[str]] = []  # 测试观测点

      def _collect_mtimes(self, root: str) -> list[tuple[str, int]]:
          from devprompt_ai.domain.retrieval.symbols import EXCLUDED_DIRS

          entries: list[tuple[str, int]] = []
          for dirpath, dirnames, filenames in os.walk(root):
              dirnames[:] = [d for d in dirnames if d not in EXCLUDED_DIRS]
              for fn in filenames:
                  fp = os.path.join(dirpath, fn)
                  try:
                      st = os.stat(fp)
                  except OSError:
                      continue
                  entries.append((os.path.relpath(fp, root), st.st_mtime_ns))
          return entries

      def _current_key(self, root: str) -> str:
          if self.mtime_digest_override is not None:
              return cache_key(root, [("override", hash(self.mtime_digest_override))])
          return cache_key(root, self._collect_mtimes(root))

      async def get(self, root: str) -> SymbolTableResult:
          now = time.monotonic()
          key = self._current_key(root)
          if (
              self._cache_symbols is not None
              and self._cache_key == key
              and (now - self._cached_at) < self._ttl
          ):
              logger.info("cache.hit", extra={"what": "symbol_table", "root": root})
              return SymbolTableResult(symbols=self._cache_symbols or [], cache_hit=True)

          args = build_ctags_args(root)
          self.calls_args.append(args)
          raw = await self._runner.run(args, timeout=30.0)  # type: ignore[attr-defined]
          symbols = parse_ctags_output(raw)
          self._cache_key = key
          self._cache_symbols = symbols
          self._cached_at = now
          logger.info("cache.miss", extra={"what": "symbol_table", "root": root})
          return SymbolTableResult(symbols=symbols, cache_hit=False)


  @dataclass(frozen=True, slots=True)
  class SymbolTableResult:
      symbols: list[Symbol]
      cache_hit: bool
  ```

  ```python
  """F-102 候选检索：原句词元 × 符号词元 打分排序，硬约束 ≤25 条（MUST-09）。

  纯函数零 I/O；候选仅符号名+路径，类型层无正文字段。
  来源通道：symbol（符号名命中）/ path（路径词元命中）。
  「similarity」通道为 F-801 不达标时的重做预案（放宽匹配/模糊匹配），首版不实现——
  按真源 §4.4 门禁处置：召回率 <80% 时才加强，避免过早优化。
  """

  from __future__ import annotations

  from dataclasses import dataclass

  from devprompt_ai.common.text import split_identifier, tokenize_sentence
  from devprompt_ai.domain.models import Candidate, Symbol

  MAX_CANDIDATES = 25  # MUST-09 硬约束：超出按匹配度截断，不报错
  MIN_CANDIDATES = 3  # F-102：少于 3 条触发人工指定流程

  _SYMBOL_WEIGHT = 2.0
  _PATH_WEIGHT = 1.0


  @dataclass(frozen=True, slots=True)
  class CandidateResult:
      candidates: list[Candidate]
      total_matched: int
      manual_required: bool


  def _score_symbol(sentence_terms: set[str], symbol: Symbol) -> tuple[float, CandidateSource] | None:
      """词元重叠打分：符号名命中权重 2、路径词元命中权重 1；零命中返回 None。"""
      name_terms = set(split_identifier(symbol.name))
      name_hits = len(name_terms & sentence_terms)

      path_terms = set(split_identifier(symbol.path))
      path_hits = len(path_terms & sentence_terms)

      score = name_hits * _SYMBOL_WEIGHT + path_hits * _PATH_WEIGHT
      if score <= 0:
          return None
      source: CandidateSource = "symbol" if name_hits > 0 else "path"
      return score, source


  def build_candidates(sentence: str, symbols: list[Symbol]) -> CandidateResult:
      """从符号表构建候选集：打分 → 降序 → 截断 25 → <3 置人工标志。"""
      sentence_terms = set(tokenize_sentence(sentence))

      scored: list[Candidate] = []
      for sym in symbols:
          hit = _score_symbol(sentence_terms, sym)
          if hit is None:
              continue
          score, source = hit
          scored.append(Candidate(name=sym.name, path=sym.path, line=sym.line, score=score, source=source))

      scored.sort(key=lambda c: (-c.score, len(c.name), c.path))

      total_matched = len(scored)
      truncated = scored[:MAX_CANDIDATES]  # 超限截断而非报错（MUST-09）
      manual_required = len(truncated) < MIN_CANDIDATES
      return CandidateResult(candidates=truncated, total_matched=total_matched, manual_required=manual_required)
  ```

  ```python
  """domain/retrieval 测试共用 Fake（无真实进程/文件系统）。"""

  from devprompt_ai.domain.retrieval.ports import CommandRunner, FileReader


  class FakeCommandRunner:
      """按「命令首词 + --route-key」路由到预置输出的假 runner。

      未命中路由时返回空串（等价 rg 无匹配、退出码 1 的语义）。
      """

      def __init__(self, routes: dict[tuple[str, str], str]) -> None:
          self._routes = routes
          self.calls: list[list[str]] = []

      async def run(
          self,
          args: list[str],
          *,
          timeout: float = 10.0,
          allowed_exit_codes: frozenset[int] = frozenset({0}),
          cwd: str | None = None,
      ) -> str:
          self.calls.append(args)
          word = args[0] if args else ""
          key = next((a.split("=", 1)[1] for a in args if a.startswith("--route-key=")), "")
          return self._routes.get((word, key), "")


  class FakeFileReader:
      def __init__(self, files: dict[str, str], existing: set[str] | None = None) -> None:
          self._files = files
          self._existing = existing if existing is not None else set(files)

      async def read(self, path: str) -> str:
          return self._files[path]

      async def exists(self, path: str) -> bool:
          return path in self._existing
  ```

  ```python
  """F-103 全文定位：rg 定位定义行号，文件内容提取选择器与设计 token，命名约定找测试。

  I/O 全部经 CommandRunner / FileReader Protocol（MUST-06）。
  测试路由键约定：rg 参数中的 --route-key=<键> 仅被测试 Fake 识别，
  LocalCommandRunner 在转发真实进程前会过滤掉它（见 3B process.py 的 run_command）。
  """

  from __future__ import annotations

  import json
  import re

  from devprompt_ai.domain.models import CodeLocation, Symbol
  from devprompt_ai.domain.retrieval.ports import CommandRunner, FileReader

  _RG_ALLOWED = frozenset({0, 1})  # 1 = 无匹配（rg 语义，正常）

  _CLASS_RE = re.compile(r"""(?:className|class)\s*=\s*["'{]([^"'}]+)["'}]""")
  _TOKEN_RE = re.compile(r"--[A-Za-z0-9][A-Za-z0-9-]*")


  def _with_route(args: list[str], key: str) -> list[str]:
      """在真实 rg 参数后追加测试路由键（LocalCommandRunner 转发前过滤）。"""
      return [*args, f"--route-key={key}"]


  async def locate_symbol(
      symbol: Symbol, runner: CommandRunner, reader: FileReader
  ) -> CodeLocation:
      """定位符号：定义行号（rg 首个精确匹配）+ 选择器/token（读文件正则提取）+ 测试文件（命名约定存在性检查）。"""
      # 1) 定义行号：rg --json 固定串搜索限定该文件
      def_args = _with_route(
          ["rg", "--json", "-n", "--no-heading", "--fixed-strings",
           symbol.name, "--glob", symbol.path, "."],
          f"def:{symbol.name}",
      )
      raw = await runner.run(def_args, allowed_exit_codes=_RG_ALLOWED)
      definition_line = _first_match_line(raw)

      # 2) 选择器与设计 token：读文件文本正则提取
      text = await reader.read(symbol.path)
      selectors: list[str] = []
      for m in _CLASS_RE.finditer(text):
          for cls in m.group(1).replace("'", " ").replace('"', " ").split():
              if cls and cls not in selectors:
                  selectors.append(cls)

      tokens: list[str] = []
      for tok in _TOKEN_RE.findall(text):
          if tok not in tokens:
              tokens.append(tok)

      # 3) 关联测试文件：按命名约定对候选路径做存在性检查
      test_paths: list[str] = []
      stem, dot, ext = symbol.path.rpartition(".")
      base = stem if dot else symbol.path
      file_name = symbol.path.rsplit("/", 1)[-1]
      ext_clean = ext if dot else "ts"
      candidates = [f"{base}.test.{ext_clean}", f"{base}.spec.{ext_clean}",
                    f"test_{file_name}", f"{base}_test.{ext_clean}"]
      for cand in candidates:
          if await reader.exists(cand):
              test_paths.append(cand)

      return CodeLocation(
          symbol_name=symbol.name,
          path=symbol.path,
          definition_line=definition_line,
          selectors=selectors,
          tokens=tokens,
          test_paths=test_paths,
      )


  def _first_match_line(rg_json: str) -> int:
      """从 rg --json 输出提取首个匹配行号；无匹配返回 0（F-104 的 symbol_in_file 会标 fail）。"""
      for line in rg_json.splitlines():
          try:
              obj = json.loads(line)
          except json.JSONDecodeError:
              continue
          if obj.get("type") == "match":
              ln = obj.get("data", {}).get("line_number")
              if isinstance(ln, int):
                  return ln
      return 0
  ```

  ```python
  """F-104 事实取证：ripgrep 逐项验证 7 项客观事实，布尔 + 证据，无「未知」。

  证据截断 200 字符（MUST-09 精神：控制进入 state 的体积）。
  I/O 全部经 CommandRunner / FileReader Protocol（MUST-06）。
  """

  from __future__ import annotations

  from devprompt_ai.domain.models import CodeLocation, FactKey, FactResult
  from devprompt_ai.domain.retrieval.ports import CommandRunner, FileReader

  _RG_ALLOWED = frozenset({0, 1})
  _EVIDENCE_LIMIT = 200

  _KNOWN_EXTS = {".tsx", ".ts", ".jsx", ".js", ".vue", ".py", ".css", ".scss", ".less", ".html"}


  def _ev(raw: str) -> str:
      raw = raw.strip().replace("\n", " ⏎ ")
      return raw[:_EVIDENCE_LIMIT] if raw else "未找到"


  async def verify_facts(
      location: CodeLocation, sentence: str, runner: CommandRunner, reader: FileReader
  ) -> list[FactResult]:
      """逐项验证七项事实（F-104 行为列）；每项独立验证，互不短路。"""
      path, name = location.path, location.symbol_name
      results: list[FactResult] = []

      # 1) 目标文件存在（文件系统检查经 FileReader）
      exists = await reader.exists(path)
      results.append(FactResult("target_file_exists", exists, f"file={path}" if exists else f"missing={path}"))

      # 2) 目标组件在文件中存在（rg 固定串搜索该文件）
      out = await runner.run(["rg", "--fixed-strings", "-n", name, "--glob", path, "."],
                             allowed_exit_codes=_RG_ALLOWED)
      sym_hit = next((l for l in out.splitlines() if name in l), "")
      results.append(FactResult("symbol_in_file", bool(sym_hit), _ev(sym_hit)))

      # 3) 样式 token 已定义（全局搜 token 名的定义处，限定样式文件）
      token = location.tokens[0] if location.tokens else ""
      tok_out = await runner.run(["rg", "--fixed-strings", "-n", token, "--glob", "*.css", "."],
                                 allowed_exit_codes=_RG_ALLOWED) if token else ""
      tok_hit = next((l for l in tok_out.splitlines() if token and token in l), "")
      results.append(FactResult("token_defined", bool(tok_hit), _ev(tok_hit)))

      # 4) 组件含 data-testid（该文件内）
      out = await runner.run(["rg", "-n", "data-testid", "--glob", path, "."], allowed_exit_codes=_RG_ALLOWED)
      tid = next((l for l in out.splitlines() if "data-testid" in l), "")
      results.append(FactResult("has_data_testid", bool(tid), _ev(tid)))

      # 5) 多状态样式（:hover / :focus / :active 任一）
      out = await runner.run(["rg", "-n", "-e", ":hover", "-e", ":focus", "-e", ":active", "--glob", path, "."],
                             allowed_exit_codes=_RG_ALLOWED)
      st = next((l for l in out.splitlines() if any(s in l for s in (":hover", ":focus", ":active"))), "")
      results.append(FactResult("has_state_styles", bool(st), _ev(st)))

      # 6) 文件类型匹配预期（后缀在已知前端/样式后缀集内）
      last_seg = path.rsplit("/", 1)[-1]
      ext = ("." + last_seg.rsplit(".", 1)[-1]) if "." in last_seg else ""
      results.append(FactResult("file_type_matches", ext in _KNOWN_EXTS, f"ext={ext or 'none'}"))

      # 7) 关联测试文件存在（F-103 已定位的 test_paths 逐一存在性检查）
      test_evidence = "未找到"
      test_ok = False
      for tp in location.test_paths:
          if await reader.exists(tp):
              test_ok, test_evidence = True, f"test={tp}"
              break
      results.append(FactResult("test_file_exists", test_ok, test_evidence))

      return results
  ```

  - 注意事项：token_defined 在「token 未从文件提取到」时证据为「未找到」且判 False——这与 F-104「无未知」一致（未提取到 token 即无法证明已定义）；rg 全库搜索以 `--glob` 限定范围、`--fixed-strings` 防正则注入（symbol.name 来自 ctags 输出）；`FactResult` 用位置参数（key/passed/evidence 与 dataclass 字段序一致）

  ```python
  """

  from __future__ import annotations

  from typing import Any


  class DevPromptError(Exception):
      """领域异常基类：code 为 §3.3 错误码，context 为结构化上下文。"""

      code: int = 9001

      def __init__(self, message: str, *, code: int | None = None,
                   context: dict[str, Any] | None = None) -> None:
          super().__init__(message)
          if code is not None:
              self.code = code
          self.context: dict[str, Any] = context or {}


  class RetrievalError(DevPromptError):
      """检索与取证失败（2001–2006）。"""

      def __init__(self, message: str, *, code: int = 2003, context: dict[str, Any] | None = None) -> None:
          super().__init__(message, code=code, context=context)


  class DecisionError(DevPromptError):
      """决策层失败（3001–3004）。"""

      def __init__(self, message: str, *, code: int = 3001, context: dict[str, Any] | None = None) -> None:
          super().__init__(message, code=code, context=context)


  class GenerationError(DevPromptError):
      """生成失败（4001–4003）。"""

      def __init__(self, message: str, *, code: int = 4001, context: dict[str, Any] | None = None) -> None:
          super().__init__(message, code=code, context=context)


  class StateError(DevPromptError):
      """状态机违规（6001–6003）。"""

      def __init__(self, message: str, *, code: int = 6001, context: dict[str, Any] | None = None) -> None:
          super().__init__(message, code=code, context=context)


  class ConfigurationError(DevPromptError):
      """配置错误（9002）。"""

      def __init__(self, message: str, context: dict[str, Any] | None = None) -> None:
  ```

  ```python
      llm_base_url: str = ""

          env_file=None, case_sensitive=False, extra="ignore"
      )

          fresh = AppConfig(_env_file=None)
          for name in type(self).model_fields:
              setattr(self, name, getattr(fresh, name))
          self.validate()
  ```

  ```python
  """F-201 决策层接口：三类能力 + 统一输出结构（形态无关，MUST-12）。

  输出用 pydantic 模型（openai-agents output_type 要求），校验规则内聚：
  排序必须是 0..n-1 的排列、置信度 ∈ [0,1]、Score ∈ [0,4]。
  """

  from __future__ import annotations

  from typing import Protocol, runtime_checkable

  from pydantic import BaseModel, Field, field_validator

  from devprompt_ai.domain.models import Candidate, FactResult


  class RankComponentsOutput(BaseModel):
      """组件排序输出：order 为候选索引（按最优在前），confidence ∈ [0,1]。"""

      order: list[int]
      confidence: float = Field(ge=0.0, le=1.0)

      @field_validator("order")
      @classmethod
      def _order_must_be_permutation(cls, v: list[int]) -> list[int]:
          if sorted(v) != list(range(len(v))):
              raise ValueError(f"order 必须是 0..n-1 的排列，实际 {v}")
          return v


  class FactJudgment(BaseModel):
      key: str
      passed: bool
      confidence: float = Field(ge=0.0, le=1.0)


  class FactProbeOutput(BaseModel):
      judgments: list[FactJudgment]
      overall_confidence: float = Field(ge=0.0, le=1.0)


  class VariantScore(BaseModel):
      variant: Literal["A", "B", "C"]
      score: int = Field(ge=0, le=4)  # F-401 五档刻度
      confidence: float = Field(ge=0.0, le=1.0)


  class ScorePromptsOutput(BaseModel):
      scores: list[VariantScore]


  @runtime_checkable
  class DecisionPort(Protocol):
      """三类决策能力（F-201 职责表）。实现：decision/fallback.py（LLM）、decision/jev.py（阶段五）。"""

      async def rank_components(self, sentence: str, candidates: list[Candidate]) -> RankComponentsOutput:
          """组件排序：原句 + 候选（仅符号名+路径，MUST-09 顺序：候选已在检索段压好）。"""
          ...

      async def probe_facts(self, sentence: str, facts: list[FactResult]) -> FactProbeOutput:
          """事实研判：取证结果 + 原句 → 各项判定 + 综合置信度。"""
          ...

      async def score_prompts(self, context: str, variants: list[tuple[str, str]]) -> ScorePromptsOutput:
          """方案打分：上下文 + [(档位, 正文)] → 三档适配度（阶段五接入）。"""
          ...
  ```

  ```python
  """多供应商 LLM 封装（F-004）：OpenAI 兼容协议 + Chat Completions + 禁 tracing。

  SDK 三约束（技术栈 §4.5）：set_default_openai_api("chat_completions")；
  set_tracing_disabled(True)；base_url 经 set_default_openai_client 注入。
  """

  from __future__ import annotations

  import asyncio
  from typing import TypeVar

  from agents import Agent, Runner, set_default_openai_api, set_default_openai_client, set_tracing_disabled
  from openai import AsyncOpenAI
  from pydantic import BaseModel

  from devprompt_ai.config import AppConfig
  from devprompt_ai.domain.errors import ConfigurationError

  ModelT = TypeVar("ModelT", bound=BaseModel)

  _client: AsyncOpenAI | None = None
  _configured_model: str = ""


  def _ensure_client(cfg: AppConfig) -> AsyncOpenAI:
      """惰性初始化并全局注册 OpenAI 兼容客户端（同一配置只注册一次）。"""
      global _client, _configured_model
      if not cfg.llm_base_url or not cfg.llm_model:
          raise ConfigurationError(
              "LLM 未配置：需要 LLM_BASE_URL / LLM_MODEL 环境变量",
              context={"llm_base_url_set": bool(cfg.llm_base_url), "llm_model_set": bool(cfg.llm_model)},
          )
      if _client is None or _configured_model != cfg.llm_model:
          _client = AsyncOpenAI(base_url=cfg.llm_base_url, api_key=cfg.llm_api_key or "EMPTY")
          set_default_openai_client(_client)
          set_default_openai_api("chat_completions")  # 第三方基本不支持 Responses API
          set_tracing_disabled(True)  # 第三方 key 走 tracing 会 401
          _configured_model = cfg.llm_model
      return _client


  async def chat_structured(
      cfg: AppConfig, output_type: type[ModelT], system: str, user: str, *, timeout: float = 10.0
  ) -> ModelT:
      """单次结构化调用：system+user → output_type 实例（MUST-13：wait_for 超时）。"""
      _ensure_client(cfg)
      agent = Agent(name="devprompt-llm", instructions=system, output_type=output_type,
                    model=cfg.llm_model)
      result = await asyncio.wait_for(Runner.run(agent, user, max_turns=1), timeout=timeout)
      output = result.final_output
      if not isinstance(output, output_type):
          from devprompt_ai.domain.errors import GenerationError

          raise GenerationError("LLM 返回结构不符合预期契约", code=4003,
                                context={"actual_type": type(output).__name__})
      return output
  ```

  ```python
  结构校验由 pydantic 模型承担（port.py）；此处做跨字段语义校验。
  """

  from __future__ import annotations
  ```

  ```python
  """LLMFallbackDecider：DecisionPort 的 LLM 实现（F-201；无 Jev 密钥时的默认实现，MUST-12）。"""

  from __future__ import annotations

  from devprompt_ai.config import AppConfig
  from devprompt_ai.decision.prompts import FACTS_SYSTEM, FACTS_USER, RANK_SYSTEM, RANK_USER, SCORE_SYSTEM, SCORE_USER
  from devprompt_ai.domain.decision.port import (
      DecisionPort,
      FactProbeOutput,
      RankComponentsOutput,
      ScorePromptsOutput,
  )
  from devprompt_ai.domain.models import Candidate, FactResult
  from devprompt_ai.harness.guardrails import validate_ranking
  from devprompt_ai.harness.llm import chat_structured


  class LLMFallbackDecider:
      """三类能力同构：单次调用 + 结构化输出；输出结构与 Jev 版完全一致（MUST-12）。"""

      def __init__(self, cfg: AppConfig) -> None:
          self._cfg = cfg

      async def rank_components(self, sentence: str, candidates: list[Candidate]) -> RankComponentsOutput:
          listing = "\n".join(f"{i}. {c.name}  ({c.path})" for i, c in enumerate(candidates))
          out = await chat_structured(
              self._cfg, RankComponentsOutput,
              RANK_SYSTEM, RANK_USER.format(sentence=sentence, candidates=listing),
          )
          validate_ranking(out, len(candidates))
          return out

      async def probe_facts(self, sentence: str, facts: list[FactResult]) -> FactProbeOutput:
          listing = "\n".join(f"- {f.key}: {'真' if f.passed else '假'}（证据：{f.evidence}）" for f in facts)
          return await chat_structured(
              self._cfg, FactProbeOutput,
              FACTS_SYSTEM, FACTS_USER.format(sentence=sentence, facts=listing),
          )

      async def score_prompts(self, context: str, variants: list[tuple[str, str]]) -> ScorePromptsOutput:
          listing = "\n\n".join(f"【方案 {key}】\n{body}" for key, body in variants)
          return await chat_structured(
              self._cfg, ScorePromptsOutput,
              SCORE_SYSTEM, SCORE_USER.format(context=context, variants=listing),
          )


  def create_decider(cfg: AppConfig) -> DecisionPort:
      """决策层工厂（MUST-12 可降级结构）：Jev 密钥存在且 decider=jev → JevDecider（阶段五实现，
      本阶段返回 LLM 版并在日志记 decider.switched 原因=jev-not-available）；否则 LLM 版。"""
      # 阶段五 T5-9 替换此工厂的 Jev 分支；本阶段只有 LLM 路径。
      return LLMFallbackDecider(cfg)
  ```

  ```python
  """F-801 第一段召回率实测（离线，不调任何外部 API）。

  用法：uv run python scripts/measure_recall.py --manifest eval/recall/manifest.json --out eval/recall
  输出：report-<日期>.json（机器可读）+ report-<日期>.md（人读报告）
  """

  from __future__ import annotations

  import argparse
  import asyncio
  import json
  from dataclasses import dataclass
  from datetime import date
  from pathlib import Path

  from devprompt_ai.domain.models import Symbol
  from devprompt_ai.domain.retrieval.candidates import build_candidates
  from devprompt_ai.harness.process import LocalCommandRunner, SymbolTableProvider


  @dataclass(frozen=True, slots=True)
  class Case:
      case_id: str
      sentence: str
      expected_symbol: str
      expected_path: str


  @dataclass(frozen=True, slots=True)
  class CaseResult:
      case_id: str
      sentence: str
      hit: bool
      candidate_count: int
      matched_rank: int | None
      top5: list[str]


  @dataclass(frozen=True, slots=True)
  class RepoResult:
      repo_id: str
      symbol_total: int
      small_repo: bool
      results: list[CaseResult]

      @property
      def hits(self) -> int:
          return sum(1 for r in self.results if r.hit)

      @property
      def recall(self) -> float:
          return self.hits / len(self.results) if self.results else 0.0


  def _load_cases(repo_dir: Path) -> list[Case]:
      cases: list[Case] = []
      with open(repo_dir / "cases.jsonl", encoding="utf-8") as f:
          for line in f:
              line = line.strip()
              if not line:
                  continue
              obj = json.loads(line)
              cases.append(Case(obj["id"], obj["sentence"], obj["expected_symbol"], obj["expected_path"]))
      return cases


  async def measure_repo(repo: dict[str, object], provider: SymbolTableProvider,
                         cases_root: Path) -> RepoResult:
      root = str(repo["path"])
      repo_id = str(repo["id"])
      table = await provider.get(root)
      symbols: list[Symbol] = table.symbols
      small = bool(repo.get("small", False)) or len(symbols) < 25
      results: list[CaseResult] = []
      for case in _load_cases(cases_root / repo_id):
          cand = build_candidates(case.sentence, symbols)
          hit_rank: int | None = None
          for i, c in enumerate(cand.candidates):
              if c.name == case.expected_symbol and c.path == case.expected_path:
                  hit_rank = i
                  break
          results.append(CaseResult(
              case_id=case.case_id,
              sentence=case.sentence,
              hit=hit_rank is not None,
              candidate_count=len(cand.candidates),
              matched_rank=hit_rank,
              top5=[f"{c.name}@{c.path}" for c in cand.candidates[:5]],
          ))
      return RepoResult(repo_id=repo_id, symbol_total=len(symbols), small_repo=small, results=results)


  def _report_md(repos: list[RepoResult], out: Path) -> Path:
      total_cases = sum(len(r.results) for r in repos)
      total_hits = sum(r.hits for r in repos)
      lines = [
          "# F-801 第一段 · 候选集召回率实测报告",
          "",
          f"- 日期：{date.today().isoformat()}　仓库数：{len(repos)}　样本数：{total_cases}",
          f"- **总召回率：{total_hits}/{total_cases} = {total_hits / total_cases:.1%}**（通过线 ≥80%）",
          "",
          "| 仓库 | 符号总数 | <25 标注 | 命中 | 召回率 |",
          "|---|---|---|---|---|",
      ]
      for r in repos:
          flag = "⚠️ 是" if r.small_repo else "否"
          lines.append(f"| {r.repo_id} | {r.symbol_total} | {flag} | {r.hits}/{len(r.results)} | {r.recall:.1%} |")
      small = [r for r in repos if r.small_repo]
      if small:
          lines += ["", "## 小仓库警告", ""]
          lines += [f"- {r.repo_id}（{r.symbol_total} 符号）：召回率虚高，单独统计为 {sum(x.hits for x in small)}/{sum(len(x.results) for x in small)}"]
      misses = [(r, c) for r in repos for c in r.results if not c.hit]
      if misses:
          lines += ["", "## 失败案例（逐条）", ""]
          for r, c in misses[:20]:
              lines.append(f"- [{r.repo_id}/{c.case_id}] 原句：{c.sentence}")
              lines.append(f"  - 期望：{c.case_id} 目标未入候选；top5：{'; '.join(c.top5)}")
      lines += ["", "## 结论", "", f"- {'✅ 达标，可进入阶段四' if total_hits / total_cases >= 0.8 else '⛔ 未达标：检索层必须重做（放宽匹配 / 加入相似度排序 / 分层召回），不得进入阶段四（MUST-25）'}"]
      report = out / f"report-{date.today().isoformat()}.md"
      report.write_text("\n".join(lines), encoding="utf-8")
      return report


  async def main_async(manifest_path: Path, out_dir: Path, cases_root: Path) -> int:
      manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
      provider = SymbolTableProvider(LocalCommandRunner())
      repos = [await measure_repo(repo, provider, cases_root) for repo in manifest["repos"]]
      total_cases = sum(len(r.results) for r in repos)
      total_hits = sum(r.hits for r in repos)
      recall = total_hits / total_cases if total_cases else 0.0

      payload = {
          "date": date.today().isoformat(),
          "recall": recall,
          "passed": recall >= 0.8,
          "repos": [
              {"id": r.repo_id, "symbol_total": r.symbol_total, "small": r.small_repo,
               "hits": r.hits, "cases": len(r.results),
               "details": [{"id": c.case_id, "hit": c.hit, "rank": c.matched_rank,
                            "sentence": c.sentence, "top5": c.top5} for c in r.results]}
              for r in repos
          ],
      }
      out_dir.mkdir(parents=True, exist_ok=True)
      json_path = out_dir / f"report-{date.today().isoformat()}.json"
      json_path.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
      md_path = _report_md(repos, out_dir)
      print(f"召回率 {total_hits}/{total_cases} = {recall:.1%}（{'通过' if payload['passed'] else '未通过'}）")
      print(f"报告：{md_path}")
      return 0 if payload["passed"] else 1


  def main() -> int:
      parser = argparse.ArgumentParser()
      parser.add_argument("--manifest", default="eval/recall/manifest.json")
      parser.add_argument("--out", default="eval/recall")
      parser.add_argument("--cases-root", default="eval/recall")
      args = parser.parse_args()
      return asyncio.run(main_async(Path(args.manifest), Path(args.out), Path(args.cases_root)))


  if __name__ == "__main__":
      raise SystemExit(main())
  ```

  - 注意事项：原句必须真实自然（含中文模糊表述）；expected 必须以 ctags 实际解析为准（避免标注了 ctags 抽不出的符号导致假阴性——此类样本剔除并记录）

- [ ] **第四步：执行实测并出报告**

  ```powershell
  uv run python scripts/measure_recall.py
  ```

    - **<80%**：按 MUST-25 处置——检索层重做（按序尝试：① 调 `_score_symbol` 权重；② 启用 similarity 通道：子序列/编辑距离模糊匹配；③ 分层召回：path 级候选池）；每轮重做后重测并保留历史报告对比；**未达标期间冻结阶段四任何任务**
  - 注意事项：报告必须含 <25 符号仓库的单独标注（脚本已实现）；全程不调 LLM/Jev（离线）

### 8.3 任务边界

| 维度 | 界定 |
|---|---|
| 范围内 | 语料契约、测量脚本、selftest、正式实测、报告、真源状态更新 |
| 范围外 | 排序准确率（阶段五 T5-10，需 DecisionPort）；重做策略的最终选择（用户）；仓库涉密评估（用户） |
| 起点 | 3C+3D+3A 绿灯 |
| 终点 | 报告产出 + 门禁判定 + 真源状态更新 + 提交 |
| 前置条件 | 3C/3D（测量主体）、3A（报告引用）、用户语料签收 |
| 后置输出 | **阶段四开工许可（或重做循环）**；失败案例清单 → 检索层加强的输入 |
| 接口标准 | 脚本退出码 0=达标/1=未达标；报告文件名 `report-<ISO日期>.json/.md` |
| 责任划分 | Agent：基建、预标注、实测、报告；用户：仓库选定、抽检签字、<80% 时的策略决策 |

| 合规要求 | 语料不入库涉密 | .gitignore | 报告与 selftest 已忽略（或用户批准入库） |
